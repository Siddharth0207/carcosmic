/** * CAR COSMIC ACADEMY — PENPOT AUTOMATION SCRIPT * Run this snippet directly in Penpot Plugin Console or Penpot MCP `execute_code`. */ (async () => { const TOKENS = { red: "#DE322D", ink: "#0A0A0A", surface: "#14151A", light: "#F3F3F3", white: "#FFFFFF" }; const SECTIONS = [ { name: "01 · Navbar & Sticky Header", h: 84, dark: true }, { name: "02 · Hero Section & Admission CTAs", h: 880, dark: true }, { name: "03 · Brand Partners Infinite Marquee", h: 116, dark: true }, { name: "04 · IDA Master Trainer Raj Deshmukh Spotlight", h: 620, dark: false }, { name: "05 · Live Studio & Supercar Highlights", h: 580, dark: false }, { name: "06 · 3D Rolling Counter Metrics", h: 520, dark: true }, { name: "07 · Comprehensive 6-Days Curriculum (01-06)", h: 860, dark: true }, { name: "08 · Hands-On Floor Time Matrix", h: 680, dark: false }, { name: "09 · Certification & Course Packages (3 Tiers)", h: 890, dark: false }, { name: "10 · FAQ Interactive Accordion", h: 640, dark: false }, { name: "11 · Verified Student Reviews & Testimonials", h: 660, dark: false }, { name: "12 · Academy Guides & Industry Journal", h: 620, dark: false }, { name: "13 · Final Enrollment Call to Action", h: 420, red: true }, { name: "14 · Comprehensive Site Footer & Newsletter", h: 540, dark: true } ]; async function buildBoard(title, width, x) { const board = penpot.createBoard(); board.name = title; board.resize(width, 7800); board.x = x; board.y = 0; board.fills = [{ fillColor: TOKENS.ink, fillOpacity: 1 }]; const flex = board.addFlexLayout(); flex.dir = "column"; for (const sec of SECTIONS) { const box = penpot.createBoard(); box.name = sec.name; box.resize(width, sec.h); box.fills = [{ fillColor: sec.red ? TOKENS.red : (sec.dark ? TOKENS.ink : TOKENS.light), fillOpacity: 1 }]; const lbl = penpot.createText(sec.name); lbl.fontSize = width > 500 ? 20 : 15; lbl.fontWeight = "600"; lbl.fills = [{ fillColor: (sec.dark || sec.red) ? TOKENS.white : TOKENS.ink, fillOpacity: 0.9 }]; box.appendChild(lbl); board.appendChild(box); } return board; } const d = await buildBoard("Desktop (1440px)", 1440, 0); const t = await buildBoard("Tablet (834px)", 834, 1540); const m = await buildBoard("Mobile (390px)", 390, 2480); if (penpot.viewport) { penpot.viewport.scrollAndZoomIntoView([d, t, m]); } return { status: "success", boards: [d.id, t.id, m.id] }; })();

1. Load via Penpot Plugin Manager

In Penpot, go to Plugins → Development → Load plugin from manifest..., select penpot/manifest.json and click Generate.

2. Execute via Penpot MCP

Connect your Penpot workspace to the MCP server and run the script via execute_code to populate all boards live.
